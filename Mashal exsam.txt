    const STORAGE_SCORES_KEY = 'retail_banking_scores_v1';
    const STORAGE_MISSED_KEY = 'retail_banking_missed_v1';

    let currentGroupKey = 'g1';
    let activeQuestions = [];
    let currentIndex = 0;
    let score = 0;
    let isAnswered = false;
    let selectedMultiIndices = [];
    let sessionWrongItems = [];
    let quizStartTime = null;

    const GROUP_NAMES = {
      g1: 'المجموعة الأولى (1 - 100)',
      g2: 'المجموعة الثانية (101 - 200)',
      g3: 'المجموعة الثالثة (201 - 302)',
      all: 'الاختبار الشامل الكامل (1 - 302)',

    };

    const ARABIC_LETTERS = ['أ', 'ب', 'ج', 'د', 'هـ', 'و'];

    function getSavedScores() {
      try {
        return JSON.parse(localStorage.getItem(STORAGE_SCORES_KEY)) || {};
      } catch (e) {
        return {};
      }
    }

    function saveGroupScore(groupKey, newScore, total) {
      if (groupKey === 'missed') return;
      try {
        const scores = getSavedScores();
        const percent = Math.round((newScore / total) * 100);
        const prev = scores[groupKey];
        if (!prev || percent >= prev.percent) {
          scores[groupKey] = { score: newScore, total, percent };
        } else {
          scores[groupKey].lastScore = newScore;
          scores[groupKey].lastPercent = percent;
        }
        localStorage.setItem(STORAGE_SCORES_KEY, JSON.stringify(scores));
      } catch (e) {}
    }

    function getMissedIds() {
      try {
        return JSON.parse(localStorage.getItem(STORAGE_MISSED_KEY)) || [];
      } catch (e) {
        return [];
      }
    }

    function addMissedId(qId) {
      try {
        const ids = new Set(getMissedIds());
        ids.add(qId);
        localStorage.setItem(STORAGE_MISSED_KEY, JSON.stringify([...ids]));
      } catch (e) {}
    }

    function removeMissedId(qId) {
      try {
        const ids = getMissedIds().filter(id => id !== qId);
        localStorage.setItem(STORAGE_MISSED_KEY, JSON.stringify(ids));
      } catch (e) {}
    }

    function clearMissedQuestions() {
      try {
        localStorage.removeItem(STORAGE_MISSED_KEY);
        refreshDashboardStats();
      } catch (e) {}
    }

    function refreshDashboardStats() {
      const scores = getSavedScores();
      ['g1', 'g2', 'g3', 'all'].forEach(key => {
        const badge = document.getElementById(`badge-${key}`);
        if (!badge) return;
        if (scores[key]) {
          badge.textContent = `أعلى درجة: ${scores[key].score}/${scores[key].total} (${scores[key].percent}%)`;
          badge.className = key === 'all'
            ? 'text-xs font-extrabold text-amber-300'
            : 'text-xs font-extrabold text-emerald-700 bg-emerald-50 px-2.5 py-0.5 rounded-full border border-emerald-200';
        } else {
          badge.textContent = 'لم يتم الاختبار';
        }
      });

      const missedIds = getMissedIds();
      document.getElementById('savedWrongCount').textContent = missedIds.length;
      document.getElementById('missedCountInline').textContent = missedIds.length;
      const missedBox = document.getElementById('missedQuestionsBox');
      if (missedIds.length > 0) {
        missedBox.classList.remove('hidden');
      } else {
        missedBox.classList.add('hidden');
      }
    }

      const isRandom = document.getElementById('globalRandomToggle').checked;

      if (isRandom) {
        label.textContent = 'عشوائي (موصى به)';
        label.className = 'text-xs text-emerald-700 font-semibold';

        label.textContent = 'تسلسلي حسب الترتيب';
        label.className = 'text-xs text-slate-600 font-semibold';
      }
    }

    // خوارزمية خلط عشوائي (Fisher-Yates)
    function shuffleArray(arr) {
      const copy = [...arr];
      for (let i = copy.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [copy[i], copy[j]] = [copy[j], copy[i]];
      }
      return copy;
    }

    // بدء الاختبار والتنقل بين الأسئلة
    // =========================================================================
    function startQuiz(groupKey, customQuestionList = null) {
      currentGroupKey = groupKey;
      let pool = [];

      if (customQuestionList) {
        pool = [...customQuestionList];
      } else if (groupKey === 'g1') {
        pool = QUESTIONS_DB.filter(q => q.id >= 1 && q.id <= 100);
      } else if (groupKey === 'g2') {
        pool = QUESTIONS_DB.filter(q => q.id >= 101 && q.id <= 200);
      } else if (groupKey === 'g3') {
        pool = QUESTIONS_DB.filter(q => q.id >= 201 && q.id <= 302);
      } else if (groupKey === 'all') {
        pool = [...QUESTIONS_DB];
      } else if (groupKey === 'missed') {
        const missedIds = new Set(getMissedIds());
        pool = QUESTIONS_DB.filter(q => missedIds.has(q.id));
      }

      if (pool.length === 0) return;

      const isRandom = document.getElementById('globalRandomToggle').checked;
      activeQuestions = isRandom ? shuffleArray(pool) : pool;

      currentIndex = 0;
      score = 0;
      sessionWrongItems = [];
      quizStartTime = Date.now();

      document.getElementById('quizGroupTitle').textContent = GROUP_NAMES[groupKey] || 'اختبار تفاعلي';
      document.getElementById('totalQCount').textContent = activeQuestions.length;
      document.getElementById('liveScore').textContent = '0';

      switchView('quizView');
      renderCurrentQuestion();
    }

    function renderCurrentQuestion() {
      isAnswered = false;
      selectedMultiIndices = [];

      const qObj = activeQuestions[currentIndex];
      const total = activeQuestions.length;

      document.getElementById('currentQIndex').textContent = currentIndex + 1;
      document.getElementById('originalQBadge').textContent = `رقم السؤال في البنك: #${qObj.id}`;
      document.getElementById('questionText').textContent = qObj.q;

      const progressPct = Math.round(((currentIndex) / total) * 100);
      document.getElementById('progressBar').style.width = `${Math.max(progressPct, 2)}%`;

      // إظهار تنبيه الاختيار المتعدد إن وجد
      const multiHint = document.getElementById('multiSelectHint');
      const multiSubmitWrapper = document.getElementById('multiSubmitWrapper');
      const submitMultiBtn = document.getElementById('submitMultiBtn');

      if (qObj.multi > 1) {
        multiHint.classList.remove('hidden');
        multiHint.textContent = qObj.multi === 2
          ? 'مطلوب اختيار إجابتين صحيحتين'
          : `مطلوب اختيار ${qObj.multi} إجابات صحيحة`;
        multiSubmitWrapper.classList.remove('hidden');
        submitMultiBtn.disabled = true;
      } else {
        multiHint.classList.add('hidden');
        multiSubmitWrapper.classList.add('hidden');
      }

      // إخفاء التغذية الراجعة وزر التالي
      document.getElementById('feedbackBox').classList.add('hidden');
      document.getElementById('nextBtnWrapper').classList.add('hidden');

      // بناء أزرار الخيارات
      const container = document.getElementById('optionsContainer');
      container.innerHTML = '';

      qObj.options.forEach((optText, idx) => {
        const btn = document.createElement('button');
        btn.type = 'button';
        btn.dataset.index = idx;
        btn.className = 'option-card w-full text-right p-4 rounded-xl border-2 border-slate-200 bg-slate-50/60 hover:bg-white hover:border-navy-800/50 flex items-start gap-3 focus:outline-none';
        btn.onclick = () => handleOptionClick(idx);

        const indicator = document.createElement('span');
        indicator.className = 'opt-badge w-7 h-7 rounded-lg bg-white border border-slate-300 text-navy-900 font-extrabold text-xs flex items-center justify-center shrink-0 mt-0.5';
        indicator.textContent = ARABIC_LETTERS[idx] || (idx + 1);

        const textSpan = document.createElement('span');
        textSpan.className = 'text-sm sm:text-base font-semibold text-slate-800 leading-snug flex-1';
        textSpan.textContent = optText;

        btn.appendChild(indicator);
        btn.appendChild(textSpan);
        container.appendChild(btn);
      });
    }

    function handleOptionClick(idx) {
      if (isAnswered) return;
      const qObj = activeQuestions[currentIndex];

      if (qObj.multi === 1) {
        evaluateAnswer([idx]);
      } else {
        // تبديل التحديد في الأسئلة متعددة الاختيارات
        const pos = selectedMultiIndices.indexOf(idx);
        if (pos > -1) {
          selectedMultiIndices.splice(pos, 1);
        } else {
          if (selectedMultiIndices.length < qObj.multi) {
            selectedMultiIndices.push(idx);
          }
        }
        updateMultiSelectionUI(qObj.multi);
      }
    }

    function updateMultiSelectionUI(requiredCount) {
      const buttons = document.querySelectorAll('#optionsContainer button');
      buttons.forEach(btn => {
        const idx = Number(btn.dataset.index);
        const badge = btn.querySelector('.opt-badge');
        if (selectedMultiIndices.includes(idx)) {
          btn.classList.remove('border-slate-200', 'bg-slate-50/60');
          btn.classList.add('border-navy-800', 'bg-blue-50/80');
          badge.classList.remove('bg-white', 'text-navy-900', 'border-slate-300');
          badge.classList.add('bg-navy-800', 'text-white', 'border-navy-800');
        } else {
          btn.classList.add('border-slate-200', 'bg-slate-50/60');
          btn.classList.remove('border-navy-800', 'bg-blue-50/80');
          badge.classList.add('bg-white', 'text-navy-900', 'border-slate-300');
          badge.classList.remove('bg-navy-800', 'text-white', 'border-navy-800');
        }
      });

      const submitBtn = document.getElementById('submitMultiBtn');
      submitBtn.disabled = selectedMultiIndices.length !== requiredCount;
    }

    function submitMultiAnswer() {
      if (isAnswered) return;
      const qObj = activeQuestions[currentIndex];
      if (selectedMultiIndices.length !== qObj.multi) return;
      evaluateAnswer([...selectedMultiIndices]);
    }

    function evaluateAnswer(userIndices) {
      isAnswered = true;
      const qObj = activeQuestions[currentIndex];
      const correctSet = new Set(qObj.correct);
      const userSet = new Set(userIndices);

      const isFullyCorrect =
        userIndices.length === qObj.correct.length &&
        userIndices.every(i => correctSet.has(i));

      if (isFullyCorrect) {
        score++;
        document.getElementById('liveScore').textContent = score;
        removeMissedId(qObj.id);
      } else {
        addMissedId(qObj.id);
        sessionWrongItems.push({
          question: qObj,
          userSelected: userIndices
        });
      }

      // تحديث ألوان الخيارات بعد الإجابة
      const buttons = document.querySelectorAll('#optionsContainer button');
      buttons.forEach(btn => {
        btn.disabled = true;
        const idx = Number(btn.dataset.index);
        const badge = btn.querySelector('.opt-badge');

        btn.classList.remove('border-slate-200', 'bg-slate-50/60', 'border-navy-800', 'bg-blue-50/80');

        if (correctSet.has(idx)) {
          btn.classList.add('border-emerald-600', 'bg-emerald-50');
          badge.className = 'opt-badge w-7 h-7 rounded-lg bg-emerald-600 text-white font-extrabold text-xs flex items-center justify-center shrink-0 mt-0.5';
          badge.textContent = '✓';
        } else if (userSet.has(idx) && !correctSet.has(idx)) {
          btn.classList.add('border-rose-600', 'bg-rose-50');
          badge.className = 'opt-badge w-7 h-7 rounded-lg bg-rose-600 text-white font-extrabold text-xs flex items-center justify-center shrink-0 mt-0.5';
          badge.textContent = '✕';
        } else {
          btn.classList.add('border-slate-200', 'bg-slate-50/40', 'opacity-60');
        }
      });

      document.getElementById('multiSubmitWrapper').classList.add('hidden');

      // عرض صندوق النتيجة والتوضيح
      const feedbackBox = document.getElementById('feedbackBox');
      const feedbackIcon = document.getElementById('feedbackIcon');
      const feedbackTitle = document.getElementById('feedbackTitle');
      const feedbackExplanation = document.getElementById('feedbackExplanation');

      feedbackBox.classList.remove('hidden', 'bg-emerald-50', 'border-emerald-300', 'text-emerald-950', 'bg-rose-50', 'border-rose-300', 'text-rose-950');

      if (isFullyCorrect) {
        feedbackBox.classList.add('bg-emerald-50', 'border-emerald-300', 'text-emerald-950');
        feedbackIcon.textContent = '✅';
        feedbackTitle.textContent = 'إجابة صحيحة! أحسنت';
      } else {
        feedbackBox.classList.add('bg-rose-50', 'border-rose-300', 'text-rose-950');
        feedbackIcon.textContent = '❌';
        const correctLabels = qObj.correct.map(i => `(${ARABIC_LETTERS[i]}) ${qObj.options[i]}`).join(' ، ');
        feedbackTitle.textContent = `إجابة خاطئة — الإجابة الصحيحة: ${correctLabels}`;
      }

      feedbackExplanation.textContent = qObj.explanation || '';

      // إظهار زر السؤال التالي أو إنهاء الاختبار
      const nextBtnWrapper = document.getElementById('nextBtnWrapper');
      const nextBtnLabel = document.getElementById('nextBtnLabel');
      nextBtnWrapper.classList.remove('hidden');

      if (currentIndex === activeQuestions.length - 1) {
        nextBtnLabel.textContent = 'عرض النتيجة النهائية';
      } else {
        nextBtnLabel.textContent = 'السؤال التالي';
      }
    }

    function nextQuestion() {
      if (currentIndex < activeQuestions.length - 1) {
        currentIndex++;
        renderCurrentQuestion();
      } else {
        finishQuiz();
      }
    }

    // =========================================================================
    // شاشة النتائج النهائية والمراجعة
    // =========================================================================
    function finishQuiz() {
      const total = activeQuestions.length;
      const percent = Math.round((score / total) * 100);
      const elapsedSeconds = Math.floor((Date.now() - quizStartTime) / 1000);
      const mins = String(Math.floor(elapsedSeconds / 60)).padStart(2, '0');
      const secs = String(elapsedSeconds % 60).padStart(2, '0');

      saveGroupScore(currentGroupKey, score, total);
      refreshDashboardStats();

      document.getElementById('resultGroupBadge').textContent = GROUP_NAMES[currentGroupKey] || 'الاختبار';
      document.getElementById('finalScoreText').textContent = `${score} / ${total}`;
      document.getElementById('finalPercentText').textContent = `${percent}%`;
      document.getElementById('finalTimeText').textContent = `${mins}:${secs}`;

      const passed = percent >= 70;
      const resultEmoji = document.getElementById('resultEmoji');
      const resultHeading = document.getElementById('resultHeading');
      const resultSubheading = document.getElementById('resultSubheading');
      const finalPercentEl = document.getElementById('finalPercentText');

      if (passed) {
        resultEmoji.textContent = '🏆';
        resultHeading.textContent = 'مبارك! اجتزت الاختبار بنجاح';
        resultSubheading.textContent = 'نتيجة مشرفة جداً وتؤهلك لدخول الاختبار الفعلي بثقة عالية.';
        finalPercentEl.className = 'text-xl sm:text-2xl font-extrabold text-emerald-600';
      } else {
        resultEmoji.textContent = '📚';
        resultHeading.textContent = 'تحتاج إلى مزيد من المراجعة';
        resultSubheading.textContent = 'نسبة الاجتياز المطلوبة 70% — راجع الأسئلة الخاطئة بالأسفل وأعد المحاولة.';
        finalPercentEl.className = 'text-xl sm:text-2xl font-extrabold text-rose-600';
      }

      // زر التدرب على أخطاء هذه الجلسة
      const retrySessionBtn = document.getElementById('retrySessionWrongBtn');
      const wrongReviewSection = document.getElementById('wrongReviewSection');
      const wrongReviewList = document.getElementById('wrongReviewList');

      if (sessionWrongItems.length > 0) {
        retrySessionBtn.classList.remove('hidden');
        document.getElementById('sessionWrongCount').textContent = sessionWrongItems.length;
        wrongReviewSection.classList.remove('hidden');
        wrongReviewList.innerHTML = '';

        sessionWrongItems.forEach((item, idx) => {
          const q = item.question;
          const userText = item.userSelected.map(i => q.options[i]).join(' + ');
          const correctText = q.correct.map(i => q.options[i]).join(' + ');

          const card = document.createElement('div');
          card.className = 'bg-white rounded-2xl p-5 shadow-sm border border-rose-200 space-y-2.5';
          card.innerHTML = `
            <div class="flex items-center justify-between text-xs text-slate-500">
              <span class="font-bold text-rose-700">خطأ رقم ${idx + 1}</span>
              <span>رقم السؤال: #${q.id}</span>
            </div>
            <p class="font-extrabold text-navy-900 text-sm sm:text-base">${q.q}</p>
            <div class="text-xs sm:text-sm space-y-1 pt-1">
              <p class="text-rose-700 bg-rose-50 px-3 py-1.5 rounded-lg"><strong>إجابتك:</strong> ${userText}</p>
              <p class="text-emerald-800 bg-emerald-50 px-3 py-1.5 rounded-lg"><strong>الإجابة الصحيحة:</strong> ${correctText}</p>
            </div>
            ${q.explanation ? `<p class="text-xs text-slate-600 pt-1">💡 ${q.explanation}</p>` : ''}
          `;
          wrongReviewList.appendChild(card);
        });
      } else {
        retrySessionBtn.classList.add('hidden');
        wrongReviewSection.classList.add('hidden');
      }

      switchView('resultsView');
    }

    function restartCurrentQuiz() {
      startQuiz(currentGroupKey);
    }

    function practiceSessionWrong() {
      if (sessionWrongItems.length === 0) return;
      const wrongQuestions = sessionWrongItems.map(item => item.question);
      startQuiz('missed', wrongQuestions);
    }

    // =========================================================================
    // التنقل بين الشاشات والنوافذ
    // =========================================================================
    function switchView(viewId) {
      ['homeView', 'quizView', 'resultsView'].forEach(id => {
        document.getElementById(id).classList.add('hidden');
      });
      document.getElementById(viewId).classList.remove('hidden');
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    function confirmExit() {
      document.getElementById('exitModal').classList.remove('hidden');
    }

    function closeExitModal() {
      document.getElementById('exitModal').classList.add('hidden');
    }

    function goHome() {
      closeExitModal();
      refreshDashboardStats();
      switchView('homeView');
    }

    // تشغيل التطبيق عند اكتمال التحميل
    window.addEventListener('DOMContentLoaded', initApp);
